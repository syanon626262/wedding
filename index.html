/** * Motion API（Google Apps Script） * * ページ本体（index.html）は GitHub Pages や Netlify に置き * このスクリプトは裏方として次を担当する * - 写真や動画のアップロード受付（Drive の resumable upload） * - 「みんなの写真」の一覧 * * アップロードの流れ * - GAS がアップロード先 URL を作ってブラウザに返し ブラウザが Drive に直接送る * この URL はそのファイル1つにしか書き込めないので アクセストークンはブラウザに渡らない * - 直接送れない環境では 8MB ずつ GAS に送り GAS が中継する */ /* ===== 設定 ===== */ const CONFIG = { // 保存先フォルダの ID（https://drive.google.com/drive/folders/ここの部分） FOLDER_ID: '1ep4soi5PT5DipmC1MCXOKjwzoGmvNBxn', // 合言葉 空文字なら入力欄を出さない PASSCODE: '', // アップロードの受付期間（日本時間）空文字なら制限なし OPEN_FROM: '', OPEN_UNTIL: '2026-10-31T23:59:59+09:00', // 1ファイルの上限サイズ MAX_FILE_BYTES: 2 * 1024 * 1024 * 1024, // 2GB // みんなの写真で一度に読み込む件数 PAGE_SIZE: 60, TIMEZONE: 'Asia/Tokyo', }; } const HANDLERS = { config: () => getConfig(), initUpload: r => initUpload(r), initUploads: r => initUploads(r), uploadChunk: r => uploadChunk(r.uploadId, r.offset, r.data), getStatus: r => getStatus(r.uploadId), list: r => listPhotos(r), getPhoto: r => getPhoto(r), }; /** * ページからは Content-Type: text/plain の POST で呼ぶ（CORS のプリフライトを避けるため） * 本文は JSON { action: 'list', ... } */ function doPost(e) { let req; try { req = JSON.parse((e && e.postData && e.postData.contents) || '{}'); } catch (err) { return json_({ ok: false, error: 'リクエストが正しくありません' }); } const handler = HANDLERS[req.action]; if (!handler) return json_({ ok: false, error: 'リクエストが正しくありません' }); try { return json_({ ok: true, result: handler(req) }); } catch (err) { console.error(req.action, err); return json_({ ok: false, error: String((err && err.message) || err) }); } } function json_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); } /* ===== 設定の取得 ===== */ function getConfig() { return { needsPasscode: !!CONFIG.PASSCODE, status: windowStatus_(), maxFileBytes: CONFIG.MAX_FILE_BYTES, }; } /* ===== アップロード ===== */ const MAX_BATCH = 20; // 1回のまとめ準備で受け付ける最大ファイル数 /** * アップロード開始 1ファイル分の resumable session を作る * @param {{name:string,size:number,mimeType:string,guestName:string,passcode:string,origin:string}} meta */ function initUpload(meta) { const r = initUploads({ files: [{ name: meta.name, size: meta.size, mimeType: meta.mimeType }], guestName: meta.guestName, passcode: meta.passcode, origin: meta.origin, }).sessions[0]; if (r.error) throw new Error(r.error); return r; } /** * 複数ファイル分の resumable session をまとめて作る（Drive への問い合わせは並列） * 1枚ずつ GAS を呼ぶより大幅に速い * @param {{files:{name:string,size:number,mimeType:string}[],guestName:string,passcode:string,origin:string}} req * @return {{sessions:({uploadId:string,uploadUrl:?string}|{error:string})[]}} */ function initUploads(req) { const status = windowStatus_(); } const files = Array.isArray(req.files) ? req.files.slice(0, MAX_BATCH) : []; if (!files.length) throw new Error('送信データが正しくありません'); const guest = clean_(req.guestName, 30) || 'ゲスト'; const stamp = Utilities.formatDate(new Date(), CONFIG.TIMEZONE, 'yyyyMMdd-HHmmss'); return; } const body = { name: `${stamp}_${guest}_${clean_(f.name, 120) || 'file'}`, parents: [CONFIG.FOLDER_ID], appProperties: { guest: guest }, }; const headers = { Authorization: 'Bearer ' + token, 'X-Upload-Content-Length': String(size) }; // HEIC などで種類が空のときは Drive に判定させる if (f.mimeType) { body.mimeType = String(f.mimeType); headers['X-Upload-Content-Type'] = String(f.mimeType); } jobs.push({ i: i, size: size, body: body, headers: headers }); }); const build = (job, withOrigin) => ({ url: 'https://www.googleapis.com/upload/drive/v3/files?uploadType=resumable&supportsAllDrives=true&fields=id,name', method: 'post', contentType: 'application/json; charset=UTF-8', headers: withOrigin ? Object.assign({ Origin: origin }, job.headers) : job.headers, payload: JSON.stringify(job.body), muteHttpExceptions: true, }); let direct = originOk; let responses = []; if (jobs.length) { try { responses = UrlFetchApp.fetchAll(jobs.map(j => build(j, direct))); } catch (e) { if (!direct) throw e; console.warn('Origin header rejected, falling back to relay', e); direct = false; responses = UrlFetchApp.fetchAll(jobs.map(j => build(j, false))); } } const cacheEntries = {}; jobs.forEach((job, k) => { const res = responses[k]; const location = header_(res, 'location'); if (res.getResponseCode() !== 200 || !location) { console.error('initUpload failed', res.getResponseCode(), res.getContentText()); sessions[job.i] = { error: 'アップロードを始められませんでした　時間をおいてお試しください' }; return; } const uploadId = Utilities.getUuid(); cacheEntries['up_' + uploadId] = JSON.stringify({ url: location, size: job.size }); sessions[job.i] = { uploadId: uploadId, uploadUrl: direct ? location : null }; }); if (Object.keys(cacheEntries).length) CacheService.getScriptCache().putAll(cacheEntries, CACHE_TTL); return { sessions: sessions, direct: direct }; } /** * 中継モード 分割したデータを1つ送る * @param {string} uploadId * @param {number} offset このチャンクの開始バイト位置 * @param {string} b64 チャンクの base64 */ function uploadChunk(uploadId, offset, b64) { const entry = getEntry_(uploadId); const bytes = Utilities.base64Decode(String(b64 || '')); const start = Number(offset); const end = start + bytes.length - 1; if (!Number.isInteger(start) || start < 0 || bytes.length === 0 || end >= entry.size) { throw new Error('送信データが正しくありません'); } if (end < entry.size - 1 && bytes.length % CHUNK_UNIT !== 0) { throw new Error('送信データが正しくありません'); } const res = UrlFetchApp.fetch(entry.url, { method: 'put', contentType: 'application/octet-stream', headers: { 'Content-Range': `bytes ${start}-${end}/${entry.size}` }, payload: bytes, muteHttpExceptions: true, followRedirects: false, }); putEntry_(uploadId, entry); // 有効期限を延ばす return parseResult_(res, uploadId); } /** 通信エラー後の再開用 Drive がどこまで受け取ったかを確認する */ function getStatus(uploadId) { const entry = getEntry_(uploadId); const res = UrlFetchApp.fetch(entry.url, { method: 'put', headers: { 'Content-Range': `bytes */${entry.size}` }, payload: '', muteHttpExceptions: true, followRedirects: false, }); return parseResult_(res, uploadId); } /* ===== みんなの写真 ===== */ /** * フォルダの写真と動画を新しい順に返す * @param {{pageToken?:string, fresh?:boolean}} req */ function listPhotos(req) { req = req || {}; const pageToken = req.pageToken ? String(req.pageToken) : ''; const cache = CacheService.getScriptCache(); if (!pageToken && !req.fresh) { const hit = cache.get('list_first'); if (hit) return JSON.parse(hit); } const params = { q: `'${CONFIG.FOLDER_ID}' in parents and trashed = false and (mimeType contains 'image/' or mimeType contains 'video/')`, orderBy: 'createdTime desc', pageSize: CONFIG.PAGE_SIZE, fields: 'nextPageToken,files(id,name,mimeType,createdTime,appProperties)', supportsAllDrives: true, includeItemsFromAllDrives: true, }; if (pageToken) params.pageToken = pageToken; const url = 'https://www.googleapis.com/drive/v3/files?' + Object.keys(params).map(k => k + '=' + encodeURIComponent(params[k])).join('&'); const res = UrlFetchApp.fetch(url, { headers: { Authorization: 'Bearer ' + ScriptApp.getOAuthToken() }, muteHttpExceptions: true, }); if (res.getResponseCode() !== 200) { console.error('list failed', res.getResponseCode(), res.getContentText()); throw new Error('写真を読み込めませんでした'); } const data = JSON.parse(res.getContentText()); const out = { items: (data.files || []).map(f => ({ id: f.id, type: String(f.mimeType).indexOf('video/') === 0 ? } catch (e) { /* 大きすぎるときはキャッシュしない */ } } return out; } /* ===== 写真の保存 ===== */ const MAX_SAVE_BYTES = 30 * 1024 * 1024; // これより大きい写真はまとめて保存の対象外 /** * 保存用に写真1枚を返す * - 写真フォルダの中のファイルだけ返す（それ以外の ID を渡されても返さない） * - 動画は対象外 * - HEIC は Google が作る原寸の JPEG に変換して返す（Android や Windows でも開けるように） * @param {{id:string}} req * @return {{name:string,mimeType:string,data:string}} data は base64 */ function getPhoto(req) { const id = String((req && req.id) || ''); if (!/^[\w-]{10,200}$/.test(id)) throw new Error('この写真は保存できません'); const meta = driveGet_(id, 'id,name,mimeType,size,parents,trashed,thumbnailLink'); if (!meta || meta.trashed || (meta.parents || []).indexOf(CONFIG.FOLDER_ID) < 0) { throw new Error('この写真は保存できません'); } const mime = String(meta.mimeType || ''); if (mime.indexOf('video/') === 0) throw new Error('動画はひとつずつ保存してください'); if (mime.indexOf('image/') !== 0) throw new Error('この写真は保存できません'); const token = ScriptApp.getOAuthToken(); const baseName = String(meta.name || 'photo').replace(/^\d{8}-\d{6}_/, ''); const isHeic = /hei[cf]/i.test(mime) || /\.hei[cf]$/i.test(baseName); if (isHeic) { const link = String(meta.thumbnailLink || ''); if (!link) throw new Error('変換できませんでした　時間をおいてお試しください'); // サムネイル URL の末尾のサイズ指定を「原寸」に変えて JPEG を受け取る const base = link.replace(/=s\d+(-[a-z0-9-]+)?$/i, ''); for (const size of ['=s0', '=s4032']) { const r = UrlFetchApp.fetch(base + size, { headers: { Authorization: 'Bearer ' + token }, muteHttpExceptions: true }); } throw new Error('変換できませんでした　時間をおいてお試しください'); } if (Number(meta.size) > MAX_SAVE_BYTES) throw new Error('大きすぎるため保存できません'); const r = UrlFetchApp.fetch( 'https://www.googleapis.com/drive/v3/files/' + encodeURIComponent(id) + '?alt=media&supportsAllDrives=true', { headers: { Authorization: 'Bearer ' + token }, muteHttpExceptions: true } ); if (r.getResponseCode() !== 200) return null; return JSON.parse(r.getContentText()); } function guestFromName_(name) { const m = String(name || '').match(/^\d{8}-\d{6}_(.+?)_/); return m ? m[1] : ''; } /* ===== 内部処理 ===== */ function parseResult_(res, uploadId) { const code = res.getResponseCode(); if (code === 200 || code === 201) { CacheService.getScriptCache().remove('up_' + uploadId); const file = JSON.parse(res.getContentText() || '{}');
